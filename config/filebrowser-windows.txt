# FileBrowser 本地 Windows 运行方式(迁移后实际部署,2026-09-06)
# 版本:filebrowser 2.63.23 windows-amd64(与 47 原容器同版本,i18n 补丁文件名 hash 才一致)
# 目录:D:\Tools\fbwin\

## 1. 启动命令(手动测试)
filebrowser.exe -r E:\NetDist -a 127.0.0.1 -p 8085 -d D:\Tools\fbwin\filebrowser.db
#   -r  网盘根 = E:\NetDist(顶层:公共/ 私人/)
#   -a  只监听 127.0.0.1(仅 frpc 回连,不暴露局域网)
#   -p  8085(与 frpc localPort 一致)
#   -d  数据库:从 47 的 /srv/fb/filebrowser.db 拷回(账号/guest scope/zh-cn locale 全保留)

## 2. 注册为 Windows 服务(NSSM,自动启动 + 崩溃 5s 重启)
nssm.exe install FileBrowserSvc "D:\Tools\fbwin\filebrowser.exe"
nssm.exe set FileBrowserSvc AppParameters "-r E:\NetDist -a 127.0.0.1 -p 8085 -d D:\Tools\fbwin\filebrowser.db"
nssm.exe set FileBrowserSvc AppDirectory "D:\Tools\fbwin"
nssm.exe set FileBrowserSvc AppStdout "D:\Tools\fbwin\logs\filebrowser-svc.out.log"
nssm.exe set FileBrowserSvc AppStderr "D:\Tools\fbwin\logs\filebrowser-svc.err.log"
nssm.exe set FileBrowserSvc AppExit Default Restart
nssm.exe set FileBrowserSvc AppRestartDelay 5000
nssm.exe set FileBrowserSvc Start SERVICE_AUTO_START

## 3. 常用操作
sc query FileBrowserSvc              # 查状态
nssm.exe restart FileBrowserSvc      # 重启
nssm.exe remove FileBrowserSvc confirm  # 卸载

## 4. 注意事项
- 网盘根结构必须保留顶层 公共/、私人/(guest scope=/公共 依赖;直接访问 scope 外返回 404)
- 目录内容更新 = 直接往 E:\NetDist 复制文件,无需任何网络操作(已无"上传到 47"环节)
- 升级 FileBrowser 后需重做 i18n 补丁(见 docs/03,hash 文件名会变)
- 电脑关机/睡眠 = 网盘下线;恢复后 NSSM 自动拉起两个服务,frpc 自动重连 47
